# Experimental host consumer; installed library contracts remain unchanged. MIT.
cmake_minimum_required(VERSION 3.20)
project(TelemetryReadManyMapping LANGUAGES CXX)
option(READ_MANY_ARM_PROBE "Build selected offline ARM roots instead of host checks" OFF)
if(READ_MANY_ARM_PROBE)
    find_package(Telemetry REQUIRED COMPONENTS core runtime encoded_server)
    foreach(mapping IN ITEMS compact service)
        string(TOUPPER "${mapping}" mode)
        add_executable(read-many-${mapping} ../../tests/read_many_mapping/ArmProbe.cpp)
        target_compile_definitions(read-many-${mapping} PRIVATE READ_MANY_PROBE_${mode}=1)
        target_compile_features(read-many-${mapping} PRIVATE cxx_std_20)
        target_link_libraries(read-many-${mapping} PRIVATE Telemetry::core Telemetry::runtime
            Telemetry::encoded_server)
        target_link_options(read-many-${mapping} PRIVATE -nostartfiles --specs=nosys.specs
            -Wl,--gc-sections -Wl,-e,read_many_${mapping}_probe
            -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/read-many-${mapping}.map)
    endforeach()
    return()
endif()
find_package(Telemetry REQUIRED COMPONENTS core runtime encoded_server resource_protocol)
set(COBS_ROOT "" CACHE PATH "External qualified COBS checkout for actual frame accounting")
if(NOT EXISTS "${COBS_ROOT}/src/cobs/Cobs.h")
    message(FATAL_ERROR "Set COBS_ROOT to an external COBS checkout")
endif()
foreach(target IN ITEMS read-many-mapping read-many-noheap)
    if(target STREQUAL "read-many-mapping")
        set(source Host.cpp)
    else()
        set(source ../../tests/read_many_mapping/NoHeap.cpp)
    endif()
    add_executable(${target} ${source} Prototype.hpp Fixture.hpp
        "${COBS_ROOT}/src/cobs/Encoder.cpp" "${COBS_ROOT}/src/cobs/Decoder.cpp")
    target_compile_features(${target} PRIVATE cxx_std_20)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_link_libraries(${target} PRIVATE Telemetry::runtime Telemetry::encoded_server
        Telemetry::core Telemetry::resource_protocol)
    target_include_directories(${target} PRIVATE "${COBS_ROOT}/src")
    get_target_property(delegate_paths Telemetry::delegate INTERFACE_INCLUDE_DIRECTORIES)
    foreach(path IN LISTS delegate_paths)
        if(path MATCHES "^[$]<BUILD_INTERFACE:(.*)>$")
            target_include_directories(${target} PRIVATE "${CMAKE_MATCH_1}/delegate")
        elseif(NOT path MATCHES "[$]<")
            target_include_directories(${target} PRIVATE "${path}/delegate")
        endif()
    endforeach()
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic-errors)
        if(target STREQUAL "read-many-noheap")
            target_compile_definitions(${target} PRIVATE READ_MANY_WRAP_ALLOCATIONS=1)
            target_link_options(${target} PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc
                -Wl,--wrap=realloc -Wl,--wrap=free)
        endif()
    endif()
endforeach()
enable_testing()
add_test(NAME read-many-mapping-contract COMMAND read-many-mapping)
add_test(NAME read-many-mapping-noheap COMMAND read-many-noheap)
