# Optional serial ReadMany example; Qt Core/SerialPort only. MIT.
cmake_minimum_required(VERSION 3.21)
project(TelemetryReadManyUart LANGUAGES CXX)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core SerialPort)
qt_standard_project_setup(REQUIRES 6.4)
get_filename_component(TELEMETRY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET Telemetry::read_many_client)
    find_package(Telemetry QUIET COMPONENTS read_many_client encoded_backend qt_serial_transport)
    if(NOT Telemetry_FOUND)
        if(TARGET Telemetry::client OR TARGET Telemetry::read_many_client)
            message(FATAL_ERROR "Installed Telemetry is incomplete; select a complete prefix or use -DCMAKE_DISABLE_FIND_PACKAGE_Telemetry=ON for this checkout")
        endif()
        set(TELEMETRY_COMPONENTS "read_many_client;encoded_backend;qt_serial_transport")
        add_subdirectory("${TELEMETRY_ROOT}" telemetry)
    endif()
endif()
foreach(_component read_many_client encoded_backend qt_serial_transport)
    if(NOT TARGET Telemetry::${_component})
        message(FATAL_ERROR "Serial ReadMany requires Telemetry::${_component}")
    endif()
endforeach()
qt_add_executable(telemetry-read-many-uart main.cpp)
target_link_libraries(telemetry-read-many-uart PRIVATE
    Telemetry::qt_serial_transport Telemetry::encoded_backend Telemetry::read_many_client)
