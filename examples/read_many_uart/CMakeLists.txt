# Optional serial ReadMany example; Qt Core/SerialPort only. MIT.
cmake_minimum_required(VERSION 3.21)
project(TelemetryReadManyUart LANGUAGES CXX)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core SerialPort)
qt_standard_project_setup(REQUIRES 6.4)
set(COBS_ROOT "" CACHE PATH "External COBS checkout")
if(NOT EXISTS "${COBS_ROOT}/src/cobs/Cobs.h")
    message(FATAL_ERROR "Set COBS_ROOT to the existing COBS checkout")
endif()
get_filename_component(TELEMETRY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET Telemetry::read_many_client)
    find_package(Telemetry QUIET COMPONENTS read_many_client encoded_backend resource_wire)
    if(NOT Telemetry_FOUND)
        if(TARGET Telemetry::client OR TARGET Telemetry::read_many_client)
            message(FATAL_ERROR "Installed Telemetry is incomplete; select a complete prefix or use -DCMAKE_DISABLE_FIND_PACKAGE_Telemetry=ON for this checkout")
        endif()
        set(TELEMETRY_COMPONENTS "read_many_client;encoded_backend;resource_wire")
        add_subdirectory("${TELEMETRY_ROOT}" telemetry)
    endif()
endif()
foreach(_component read_many_client encoded_backend resource_wire)
    if(NOT TARGET Telemetry::${_component})
        message(FATAL_ERROR "Serial ReadMany requires Telemetry::${_component}")
    endif()
endforeach()
add_library(read_many_uart_transport STATIC
    ../inspector_qml/SerialTransport.cpp ../inspector_qml/SerialTransport.hpp
    ../inspector_qml/DeviceTransport.hpp ../inspector_qml/DescriptorBootstrap.hpp
    "${COBS_ROOT}/src/cobs/Encoder.cpp" "${COBS_ROOT}/src/cobs/Decoder.cpp")
target_compile_features(read_many_uart_transport PUBLIC cxx_std_20)
target_compile_definitions(read_many_uart_transport PUBLIC QT_NO_KEYWORDS QT_NO_FOREACH)
target_include_directories(read_many_uart_transport PUBLIC ../inspector_qml "${COBS_ROOT}/src")
target_link_libraries(read_many_uart_transport PUBLIC Qt6::Core Qt6::SerialPort
    Telemetry::encoded_backend Telemetry::read_many_client Telemetry::resource_wire)
get_target_property(_delegate_paths Telemetry::delegate INTERFACE_INCLUDE_DIRECTORIES)
foreach(_path IN LISTS _delegate_paths)
    if(_path MATCHES "^[$]<BUILD_INTERFACE:(.*)>$")
        target_include_directories(read_many_uart_transport PUBLIC "${CMAKE_MATCH_1}/delegate")
    elseif(NOT _path MATCHES "[$]<")
        target_include_directories(read_many_uart_transport PUBLIC "${_path}/delegate")
    endif()
endforeach()
qt_add_executable(telemetry-read-many-uart main.cpp)
target_link_libraries(telemetry-read-many-uart PRIVATE read_many_uart_transport)
