cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(TelemetryClient LANGUAGES CXX)
endif()
option(TELEMETRY_CLIENT_RPC "Build the native one-mailbox RPC client" OFF)
if(NOT TARGET telemetry_client)
    set(TELEMETRY_COMPONENTS client)
    if(TELEMETRY_CLIENT_RPC)
        list(APPEND TELEMETRY_COMPONENTS rpc_client)
    endif()
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." "${CMAKE_CURRENT_BINARY_DIR}/modules")
elseif(TELEMETRY_CLIENT_RPC AND NOT TARGET telemetry_rpc_client)
    message(FATAL_ERROR "Select rpc_client in the parent TELEMETRY_COMPONENTS before adding the client entry point")
endif()
