cmake_minimum_required(VERSION 3.21)
project(telemetry_inspector_serial_checks LANGUAGES CXX)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network SerialPort Test)
qt_standard_project_setup(REQUIRES 6.4)
get_filename_component(TELEMETRY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(TELEMETRY_UX_ROOT "" CACHE PATH "External telemetry-ux checkout containing apps/inspector")
if(NOT EXISTS "${TELEMETRY_UX_ROOT}/apps/inspector/CMakeLists.txt")
    message(FATAL_ERROR "Set TELEMETRY_UX_ROOT to the external telemetry-ux checkout")
endif()
if(NOT EXISTS "${COBS_ROOT}/src/cobs/Cobs.h")
    message(FATAL_ERROR "Set COBS_ROOT to the explicit test-peer COBS checkout")
endif()
set(BUILD_TESTING OFF CACHE BOOL "Build only maintained serial checks" FORCE)
set(INSPECTOR_ENABLE_SERIAL ON CACHE BOOL "Exercise the actual serial port" FORCE)
# Resolve imported targets in this parent scope before Qt's deferred finalization.
find_package(Telemetry 0.1.0 EXACT CONFIG REQUIRED COMPONENTS inspector workbench
    read_many_workbench encoded_backend encoded_server core tcp_server resource_adapter qt_serial_transport)
add_subdirectory("${TELEMETRY_UX_ROOT}/apps/inspector" example)
set_target_properties(telemetry-inspector-qml PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/example")
qt_add_executable(telemetry_inspector_serial_check main.cpp)
target_compile_features(telemetry_inspector_serial_check PRIVATE cxx_std_20)
target_link_libraries(telemetry_inspector_serial_check PRIVATE inspector_qml_support Telemetry::resource_adapter Qt6::Gui Qt6::Test)
target_include_directories(telemetry_inspector_serial_check PRIVATE "${COBS_ROOT}/src")
get_target_property(_delegate_paths Telemetry::delegate INTERFACE_INCLUDE_DIRECTORIES)
foreach(_delegate_path IN LISTS _delegate_paths)
    if(_delegate_path MATCHES "^[$]<BUILD_INTERFACE:(.*)>$")
        target_include_directories(telemetry_inspector_serial_check PRIVATE "${CMAKE_MATCH_1}/delegate")
    elseif(NOT _delegate_path MATCHES "[$]<")
        target_include_directories(telemetry_inspector_serial_check PRIVATE "${_delegate_path}/delegate")
    endif()
endforeach()
if(UNIX)
    target_link_libraries(telemetry_inspector_serial_check PRIVATE util)
endif()
qt_add_executable(telemetry_inspector_serial_watchlist_check Watchlist.cpp)
target_compile_features(telemetry_inspector_serial_watchlist_check PRIVATE cxx_std_20)
target_compile_definitions(telemetry_inspector_serial_watchlist_check PRIVATE TELEMETRY_NUCLEO_READ_MANY=1)
target_link_libraries(telemetry_inspector_serial_watchlist_check PRIVATE inspector_qml_support Telemetry::resource_adapter Qt6::Gui Qt6::Test)
target_include_directories(telemetry_inspector_serial_watchlist_check PRIVATE "${COBS_ROOT}/src")
target_include_directories(telemetry_inspector_serial_watchlist_check PRIVATE ${_delegate_paths})
foreach(_delegate_path IN LISTS _delegate_paths)
    if(_delegate_path MATCHES "^[$]<BUILD_INTERFACE:(.*)>$")
        target_include_directories(telemetry_inspector_serial_watchlist_check PRIVATE "${CMAKE_MATCH_1}/delegate")
    elseif(NOT _delegate_path MATCHES "[$]<")
        target_include_directories(telemetry_inspector_serial_watchlist_check PRIVATE "${_delegate_path}/delegate")
    endif()
endforeach()
if(UNIX)
    target_link_libraries(telemetry_inspector_serial_watchlist_check PRIVATE util)
endif()
