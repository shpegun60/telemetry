# Real serial CLI, board Protocol, COBS and PTYs; no QML runtime. MIT.
cmake_minimum_required(VERSION 3.21)
project(TelemetryReadManySerialChecks LANGUAGES CXX)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core SerialPort)
qt_standard_project_setup(REQUIRES 6.4)
get_filename_component(TELEMETRY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BUILD_TESTING OFF CACHE BOOL "Only maintained serial checks" FORCE)
set(TELEMETRY_COMPONENTS "read_many_client;encoded_backend;encoded_server;core;resource_adapter;resource_protocol")
add_subdirectory("${TELEMETRY_ROOT}" telemetry)
add_subdirectory("${TELEMETRY_ROOT}/examples/read_many_uart" example)
qt_add_executable(read_many_bootstrap_check Bootstrap.cpp)
target_link_libraries(read_many_bootstrap_check PRIVATE read_many_uart_transport)
qt_add_executable(read_many_serial_check main.cpp)
target_compile_definitions(read_many_serial_check PRIVATE TELEMETRY_NUCLEO_READ_MANY=1)
target_link_libraries(read_many_serial_check PRIVATE read_many_uart_transport
    Telemetry::core Telemetry::encoded_server Telemetry::resource_adapter Telemetry::resource_protocol)
if(UNIX)
    target_link_libraries(read_many_serial_check PRIVATE util)
endif()
