# Real installed SDK consumers; no source include root fallback. MIT.
cmake_minimum_required(VERSION 3.20)
project(TelemetryStaticReadManyTests LANGUAGES CXX)
set(STATIC_MANY_GENERATED "" CACHE PATH "External generated cpp directory")
option(STATIC_MANY_PRODUCER_ONLY "Build the host Descriptor producer" OFF)
option(STATIC_MANY_STANDALONE_ONLY "Build only the minimal SDK consumer" OFF)
option(STATIC_MANY_ARM "Build the offline Cortex-M7 root" OFF)
if(STATIC_MANY_PRODUCER_ONLY)
    find_package(Telemetry REQUIRED COMPONENTS core)
    add_executable(static-many-descriptor Descriptor.cpp)
    target_link_libraries(static-many-descriptor PRIVATE Telemetry::core)
    return()
endif()
if(NOT EXISTS "${STATIC_MANY_GENERATED}/static_many_api/ClientBindings.hpp")
    message(FATAL_ERROR "STATIC_MANY_GENERATED must contain the actual generated SDK")
endif()
find_package(Telemetry REQUIRED COMPONENTS read_many_static_client encoded_backend)
if(STATIC_MANY_ARM)
    add_executable(static-many-arm Arm.cpp)
    target_include_directories(static-many-arm PRIVATE "${STATIC_MANY_GENERATED}")
    target_link_libraries(static-many-arm PRIVATE Telemetry::read_many_static_client Telemetry::encoded_backend)
    target_link_options(static-many-arm PRIVATE -nostartfiles --specs=nosys.specs
        -Wl,--gc-sections -Wl,-e,static_many_entry
        -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/static-many-arm.map)
    return()
endif()
add_executable(static-many-standalone Standalone.cpp)
target_include_directories(static-many-standalone PRIVATE "${STATIC_MANY_GENERATED}")
target_link_libraries(static-many-standalone PRIVATE Telemetry::read_many_static_client Telemetry::encoded_backend)
if(STATIC_MANY_STANDALONE_ONLY)
    return()
endif()
find_package(Telemetry REQUIRED COMPONENTS core encoded_server)
foreach(name IN ITEMS Host NoHeap Example)
    add_executable(static-many-${name} ${name}.cpp)
    target_include_directories(static-many-${name} PRIVATE "${STATIC_MANY_GENERATED}")
    target_link_libraries(static-many-${name} PRIVATE Telemetry::read_many_static_client
        Telemetry::encoded_backend Telemetry::core Telemetry::encoded_server)
endforeach()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_definitions(static-many-NoHeap PRIVATE STATIC_MANY_WRAP_ALLOCATIONS=1)
    target_link_options(static-many-NoHeap PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc
        -Wl,--wrap=realloc -Wl,--wrap=free)
endif()
