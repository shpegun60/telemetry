# Real installed SDK consumers; no source include root fallback. MIT.
cmake_minimum_required(VERSION 3.20)
project(TelemetryStaticBatchTests LANGUAGES CXX)
set(STATIC_BATCH_GENERATED "" CACHE PATH "External generated cpp directory")
option(STATIC_BATCH_PRODUCER_ONLY "Build the host Descriptor producer" OFF)
option(STATIC_BATCH_STANDALONE_ONLY "Build only the minimal SDK consumer" OFF)
option(STATIC_BATCH_ARM "Build the offline Cortex-M7 root" OFF)
if(STATIC_BATCH_PRODUCER_ONLY)
    find_package(Telemetry REQUIRED COMPONENTS core)
    add_executable(static-batch-descriptor Descriptor.cpp)
    target_link_libraries(static-batch-descriptor PRIVATE Telemetry::core)
    return()
endif()
if(NOT EXISTS "${STATIC_BATCH_GENERATED}/static_batch_api/ClientBindings.hpp")
    message(FATAL_ERROR "STATIC_BATCH_GENERATED must contain the actual generated SDK")
endif()
find_package(Telemetry REQUIRED COMPONENTS batch_static_client encoded_backend)
if(STATIC_BATCH_ARM)
    add_executable(static-batch-arm Arm.cpp)
    target_include_directories(static-batch-arm PRIVATE "${STATIC_BATCH_GENERATED}")
    target_link_libraries(static-batch-arm PRIVATE Telemetry::batch_static_client Telemetry::encoded_backend)
    target_link_options(static-batch-arm PRIVATE -nostartfiles --specs=nosys.specs
        -Wl,--gc-sections -Wl,-e,static_batch_entry
        -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/static-batch-arm.map)
    return()
endif()
add_executable(static-batch-standalone Standalone.cpp)
target_include_directories(static-batch-standalone PRIVATE "${STATIC_BATCH_GENERATED}")
target_link_libraries(static-batch-standalone PRIVATE Telemetry::batch_static_client Telemetry::encoded_backend)
if(STATIC_BATCH_STANDALONE_ONLY)
    return()
endif()
find_package(Telemetry REQUIRED COMPONENTS core batch_server)
foreach(name IN ITEMS Host NoHeap Example)
    add_executable(static-batch-${name} ${name}.cpp)
    target_include_directories(static-batch-${name} PRIVATE "${STATIC_BATCH_GENERATED}")
    target_link_libraries(static-batch-${name} PRIVATE Telemetry::batch_static_client
        Telemetry::encoded_backend Telemetry::core Telemetry::batch_server)
endforeach()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_definitions(static-batch-NoHeap PRIVATE STATIC_BATCH_WRAP_ALLOCATIONS=1)
    target_link_options(static-batch-NoHeap PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc
        -Wl,--wrap=realloc -Wl,--wrap=free)
endif()
