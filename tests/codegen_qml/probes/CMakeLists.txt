cmake_minimum_required(VERSION 3.21)
project(TelemetryQmlValueProbes LANGUAGES CXX)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Qml)
if(NOT DEFINED GENERATED_SDK)
    message(FATAL_ERROR "GENERATED_SDK must name the external explicit fixture output")
endif()
foreach(probe IN ITEMS values snapshot)
    if(probe STREQUAL "values")
        set(probe_source Probes.cpp)
    else()
        set(probe_source Snapshot.cpp)
    endif()
    add_executable(qml_${probe}_probe ${probe_source})
    target_compile_features(qml_${probe}_probe PRIVATE cxx_std_20)
    target_compile_definitions(qml_${probe}_probe PRIVATE QT_NO_KEYWORDS QT_NO_FOREACH)
    target_include_directories(qml_${probe}_probe PRIVATE "${GENERATED_SDK}" "${GENERATED_SDK}/cpp")
    target_link_libraries(qml_${probe}_probe PRIVATE Qt6::Core)
    if(probe STREQUAL "snapshot")
        target_link_libraries(qml_${probe}_probe PRIVATE Qt6::Qml)
    endif()
    if(MSVC)
        target_compile_options(qml_${probe}_probe PRIVATE /W4 /WX)
    else()
        target_compile_options(qml_${probe}_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
if(NOT DEFINED GENERATED_MODULES)
    message(FATAL_ERROR "GENERATED_MODULES must name the external empty/hostile SDK variants")
endif()
# These modules include their generated standalone codec only. Compilation of
# moc and qmltyperegistrar output requires no Telemetry, Resource or RPC headers.
foreach(module IN ITEMS empty hostile)
    add_subdirectory("${GENERATED_MODULES}/${module}/qml" "module-${module}")
endforeach()
# A real Qt consumer retains public keyword macros and receives no QT_NO_*
# definitions merely by linking the generated module. Its includes also place
# hostile DTO namespaces before the QObject declarations and moc output.
add_executable(qml_legacy_probe Legacy.cpp)
set_target_properties(qml_legacy_probe PROPERTIES AUTOMOC ON)
target_compile_features(qml_legacy_probe PRIVATE cxx_std_20)
target_link_libraries(qml_legacy_probe PRIVATE telemetry_qml_variant__hostile Qt6::Core)
if(MSVC)
    target_compile_options(qml_legacy_probe PRIVATE /W4 /WX)
else()
    target_compile_options(qml_legacy_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
